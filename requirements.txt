# of this file rather than repeating it (.github/workflows/ci.yml), so the
# board is built with the config that was checked and the two cannot drift.
numpy~=2.5
scipy~=1.18
PyYAML~=6.0
esphome==2026.9.0
# esphome pins aioesphomeapi EXACTLY (2026.9.0 wants ==46.3.0), so the version

# floor, not a range: `~=46.3` made Dependabot offer ~=46.4 the day it shipped
# (PR #38), which pip can never resolve against esphome's pin. requirements.lock
# records the version that actually installs.
aioesphomeapi>=46.4.1
